Supabase SQL Schema & Keamanan (schema.sql)

Salin skrip SQL ini ke SQL Editor di Supabase untuk menyiapkan tabel database.

-- ====================================================================
-- SENSASI MADRASAH - SUPABASE SQL SCHEMA (NIS SEBAGAI BARCODE)
-- ====================================================================

CREATE TABLE IF NOT EXISTS classes (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

CREATE TABLE IF NOT EXISTS students (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    nis VARCHAR(50) NOT NULL UNIQUE,
    name VARCHAR(150) NOT NULL,
    class_name VARCHAR(100) NOT NULL,
    gender VARCHAR(10) CHECK (gender IN ('L', 'P')),
    photo TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

CREATE TABLE IF NOT EXISTS attendance (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    student_id UUID REFERENCES students(id) ON DELETE CASCADE,
    nis VARCHAR(50) NOT NULL,
    name VARCHAR(150) NOT NULL,
    class_name VARCHAR(100) NOT NULL,
    status VARCHAR(50) DEFAULT 'Hadir',
    type VARCHAR(20) DEFAULT 'Masuk',
    timestamp TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

ALTER TABLE classes ENABLE ROW LEVEL SECURITY;
ALTER TABLE students ENABLE ROW LEVEL SECURITY;
ALTER TABLE attendance ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Public Access Classes" ON classes FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Public Access Students" ON students FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Public Access Attendance" ON attendance FOR ALL USING (true) WITH CHECK (true);

Langkah Deploy (Zero-Installation)

Panduan mengunggah aplikasi absensi Barcode Madrasah ke GitHub Pages & Supabase.

1

Buat Repository GitHub

Buat repository publik baru di GitHub (misal: sensasi-madrasah).

2

Upload File index.html

Simpan seluruh kode ini ke dalam satu file bernama index.html dan upload langsung melalui web browser GitHub.

3

Aktifkan GitHub Pages

Buka menu Settings > Pages di repository Anda, lalu pilih Branch main / root untuk langsung online.
